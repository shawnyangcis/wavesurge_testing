import { useEffect, useState } from 'react';
import { api } from './api.ts';
import type { Campaign } from '../shared/campaign.ts';
type Plan = Omit<Campaign, 'groups'> & { status: string; matched: number; requested: number; qualifyingUnique: number; reportUrl: string; groups: (Campaign['groups'][number] & { matched: number; eligible: number; taskId?: string })[] };
const statuses: Record<string, string> = { draft: '待确认方案', running: '研究中', needs_input: '需要协助', completed: '配额已满足', partial: '部分完成' };
export function Campaigns({ onClose }: { onClose: () => void }) {
  const [plans, setPlans] = useState<Plan[]>([]); const [selected, setSelected] = useState<Plan>(); const [error, setError] = useState('');
  async function refresh() { try { const facts = await api<{ id: string }[]>('/campaigns'); const values = await Promise.all(facts.map(fact => api<Plan>(`/campaigns/${encodeURIComponent(fact.id)}`))); setPlans(values); setSelected(current => values.find(value => value.id === current?.id)); } catch (e) { setError((e as Error).message); } }
  useEffect(() => { void refresh(); const timer = setInterval(refresh, 3000); return () => clearInterval(timer); }, []);
  async function start(plan: Plan) { setError(''); try { await api(`/campaigns/${encodeURIComponent(plan.id)}/start`, { revision: plan.revision }); await refresh(); } catch (e) { setError((e as Error).message); } }
  return <div className="scrim"><section className="inspector" role="dialog" aria-modal="true" aria-label="Campaign 方案"><div className="inspector-heading"><div><span className="eyebrow">CAMPAIGN / STRATEGY</span><h2>Campaign 方案</h2></div><button className="quiet" onClick={onClose}>关闭</button></div><p className="muted">在对话中描述产品、受众和目标，助手会与你共同确定创作者类型、范围和配额。</p>{error && <p role="alert">{error}</p>}{plans.length === 0 && <p>暂无方案。先在对话中描述你的 Campaign 目标。</p>}{plans.map(plan => <article className="task" key={plan.id}><span className="task-status">{statuses[plan.status]}</span><h3>{plan.title} · v{plan.revision}</h3><p>{plan.goal}</p><p>去重入选 {plan.matched}/{plan.requested}</p><button className="quiet" onClick={() => setSelected(plan)}>查看方案与配额</button></article>)}{selected && <div className="campaign-detail"><h3>{selected.title} · v{selected.revision}</h3><p>受众：{selected.audience}</p><p>同一平台账号只占一个类型配额；没有可靠跨平台身份时不猜测合并。</p>{selected.groups.map(group => <article className="finding" key={group.id}><h3>{group.spec.title} · {group.matched}/{group.spec.count}</h3><p>{group.rationale}</p><p>{group.spec.platforms.join('、')} · {group.spec.regions.length ? group.spec.regions.join('、') : '地区不限'} · {group.spec.minFollowers}–{group.spec.maxFollowers} 粉丝</p><p>候选上限 {group.spec.candidateBudget} · 合格账号池 {group.eligible}</p><ul>{group.spec.criteria.map(criterion => <li key={criterion.id}>{criterion.required ? '必选' : '偏好'}：{criterion.label}（{criterion.basis}）</li>)}</ul>{group.taskId && <a href={`/api/reports/${encodeURIComponent(group.taskId)}`} target="_blank" rel="noreferrer">查看此类核验报告</a>}</article>)}{selected.status === 'draft' && <button onClick={() => void start(selected)}>确认此方案并开始研究</button>}<p><a href={selected.reportUrl} target="_blank" rel="noreferrer">查看去重汇总报告</a></p></div>}</section></div>;
}
