export function escapeHtml(text: string) { return text.replaceAll('&', '&amp;').replaceAll('<', '&lt;').replaceAll('>', '&gt;').replaceAll('"', '&quot;').replaceAll("'", '&#39;'); }
export function safeLink(value: string) { return /^https?:\/\//i.test(value) ? escapeHtml(value) : '#'; }
function cell(value: string) { const safe = /^[=+@-]/.test(value) ? `'${value}` : value; return `"${safe.replaceAll('"', '""')}"`; }

export function evidenceReport(result: Result, image?: (id: string) => string) {
  return `<details open><summary>完整证据映射与截图</summary>${result.creator.evidence.map(evidence => {
    let screenshot = '';
    if (evidence.assetId) { try { const data = image?.(evidence.assetId); if (data?.startsWith('data:image/png;base64,')) screenshot = `<img style="max-width:100%" alt="截图证据 ${escapeHtml(evidence.id)}" src="${data}">`; } catch { screenshot = '<p>截图暂不可读取，请在工作室核对该证据编号。</p>'; } }
    return `<section><h4>${escapeHtml(evidence.id)} · ${escapeHtml(evidence.kind)}</h4><a href="${safeLink(evidence.url)}">原始来源</a><p>提供方：${escapeHtml(evidence.provider)} · 捕获：${new Date(evidence.capturedAt).toISOString()} · 发布时间：${escapeHtml(evidence.publishedAt ?? '日期未知')}${evidence.assetId ? ` · 截图编号：${escapeHtml(evidence.assetId)}` : ''}</p><pre style="white-space:pre-wrap">${escapeHtml(evidence.text)}</pre>${screenshot}</section>`;
  }).join('')}</details>`;
}
