import type { FastifyInstance } from 'fastify';
import type { CampaignService } from './campaign.ts';
import type { Campaign } from '../shared/campaign.ts';
import { escapeHtml, safeLink, evidenceReport } from './reports.ts';
export function registerCampaigns(app: FastifyInstance, service: CampaignService, image?: (id: string) => string) {
  app.get('/api/campaigns', async () => service.facts());
  app.get<{ Params: { id: string } }>('/api/campaigns/:id', async (request, reply) => { const campaign = service.store.get<Campaign>('campaign', request.params.id); if (!campaign) return reply.code(404).send({ error: 'Campaign 不存在' }); return service.summary(campaign); });
  app.post<{ Params: { id: string }; Body: { revision: number } }>('/api/campaigns/:id/start', { schema: { body: { type: 'object', required: ['revision'], additionalProperties: false, properties: { revision: { type: 'integer', minimum: 1 } } } } }, async (request, reply) => { try { return service.start(request.params.id, request.body.revision); } catch (error) { return reply.code(409).send({ error: (error as Error).message }); } });
  app.get<{ Params: { id: string } }>('/api/campaigns/:id/report', async (request, reply) => {
    const campaign = service.store.get<Campaign>('campaign', request.params.id); if (!campaign) return reply.code(404).send({ error: 'Campaign 不存在' }); const summary = service.summary(campaign);
    reply.header('Content-Security-Policy', "default-src 'none'; style-src 'unsafe-inline'; img-src data:"); return reply.type('text/html; charset=utf-8').send(`<!doctype html><html lang="zh"><meta charset="utf-8"><title>${escapeHtml(campaign.title)}</title><style>body{max-width:960px;margin:48px auto;font:16px/1.8 sans-serif;padding:24px;color:#172f2d}article{border-bottom:1px solid #ddd;padding:24px 0}a{color:#116653}</style><h1>${escapeHtml(campaign.title)}</h1><p>${escapeHtml(campaign.goal)}</p><p>受众：${escapeHtml(campaign.audience)}</p><p>方案 v${campaign.revision} · ${summary.status} · 去重入选 ${summary.matched}/${summary.requested}，合格账号池 ${summary.qualifyingUnique}。同一平台账号只占一个类型配额；跨平台同一人缺乏可靠关联时不猜测合并。</p>${summary.groups.map(group => `<article><h2>${escapeHtml(group.spec.title)} · ${group.matched}/${group.spec.count}</h2><p>${escapeHtml(group.rationale)}</p><pre>${escapeHtml(JSON.stringify(group.spec, null, 2))}</pre>${summary.selections.filter(selection => selection.groupId === group.id).map(selection => { const result = summary.results.find(item => item.id === selection.resultId)!; return `<p><a href="${safeLink(result.creator.profileUrl)}">${escapeHtml(result.creator.displayName)} @${escapeHtml(result.creator.handle)}</a></p>${evidenceReport(result, image)}`; }).join('')}${group.taskId ? `<a href="/api/reports/${encodeURIComponent(group.taskId)}">逐条件证据与未入选结果</a>` : ''}</article>`).join('')}</html>`);
  });
}
